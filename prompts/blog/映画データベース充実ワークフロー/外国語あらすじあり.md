{{(() => {
  let tmdb = {};
  try { tmdb = $('TMDb検索').item?.json || {}; } catch(e){}
  let source = {};
  try { source = $('Loop Over Items').item?.json || {}; } catch(e){}
  const list = tmdb.results || tmdb.movie_results || (tmdb.id ? [tmdb] : []);
  if (list.length === 0) return false;
  let matched = list.find(m => (m.original_language === source.target_lang) || (m.origin_country && m.origin_country.includes(source.target_country))) || list[0];
  
  let overview = '';
  let sourceLanguage = 'none';

  if (matched?.overview && matched.overview.trim().length > 0) {
    overview = matched.overview;
    sourceLanguage = matched.original_language || 'original';
  } else {
    const enTrans = matched?.translations?.translations?.find(t => t.iso_639_1 === 'en');
    if (enTrans?.data?.overview && enTrans.data.overview.trim().length > 0) {
      overview = enTrans.data.overview;
      sourceLanguage = 'en (translation)';
    } else {
      const fallbackTrans = matched?.translations?.translations?.find(t => t.data?.overview && t.data.overview.trim().length > 0);
      if (fallbackTrans) {
        overview = fallbackTrans.data.overview;
        sourceLanguage = `${fallbackTrans.iso_639_1} (translation)`;
      }
    }
  }

  // サーバーログ等での確認用（環境によって出力されます）
  try {
    console.log(`[Overview Language Check] Language: ${sourceLanguage}, Length: ${overview.trim().length}`);
  } catch(e){}

  return overview.trim().length > 0;
})()}}
